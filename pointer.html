Question 1.

Write a C program to enter and print all elements of an array using a pointer.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a[5], i;
    int *p;

    clrscr();

    p = a;

    printf("Enter 5 elements:\n");

    for (i = 0; i < 5; i++)
    {
        scanf("%d", p + i);
    }

    printf("Array elements are:\n");

    for (i = 0; i < 5; i++)
    {
        printf("%d ", *(p + i));
    }

    getch();
}

Sample Output

Enter 5 elements:
10
20
30
40
50

Array elements are:
10 20 30 40 50

Question 2.

Write a program to swap two numbers using pointers.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a, b, temp;
    int *p, *q;

    clrscr();

    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);

    p = &a;
    q = &b;

    temp = *p;
    *p = *q;
    *q = temp;

    printf("After swapping:\n");
    printf("A = %d\n", a);
    printf("B = %d", b);

    getch();
}

Sample Output

Enter two numbers: 10 20

After swapping:
A = 20
B = 10

Question 3.

Write a program in which we are having 2D array of 2 by 3. Insert the value through pointer and print the values.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a[2][3], i, j;
    int *p;

    clrscr();

    p = &a[0][0];

    printf("Enter 6 elements:\n");

    for (i = 0; i < 6; i++)
    {
        scanf("%d", p + i);
    }

    printf("Array elements are:\n");

    for (i = 0; i < 2; i++)
    {
        for (j = 0; j < 3; j++)
        {
            printf("%d ", *(p + i * 3 + j));
        }

        printf("\n");
    }

    getch();
}

Sample Output

Enter 6 elements:
10 20 30 40 50 60

Array elements are:
10 20 30
40 50 60
