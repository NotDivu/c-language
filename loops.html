Question 1.

Enter a number from the user and find out the binary of that number?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int num, rem, i=0;
    int a[32];

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    if(num==0)
    {
        printf("Binary = 0");
    }
    else
    {
        while(num>0)
        {
            rem=num%2;
            a[i]=rem;
            num=num/2;
            i++;
        }

        printf("Binary = ");

        for(i=i-1; i>=0; i--)
        {
            printf("%d", a[i]);
        }
    }

    getch();
}

Sample Output

Enter a number: 10
Binary = 1010

Question 12.

Enter a choice from the user. If choice is 1 perform addition, 2 subtraction, 3 multiplication, 4 division, 5 modulo. For any other choice print invalid. Enter Q to quit the program.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a, b;
    char ch;

    clrscr();

    while(1)
    {
        printf("\n1. Addition");
        printf("\n2. Subtraction");
        printf("\n3. Multiplication");
        printf("\n4. Division");
        printf("\n5. Modulo");
        printf("\nEnter choice or Q to quit: ");

        scanf(" %c", &ch);

        if(ch=='Q' || ch=='q')
        {
            break;
        }

        if(ch>='1' && ch<='5')
        {
            printf("Enter two numbers: ");
            scanf("%d %d", &a, &b);

            switch(ch)
            {
                case '1':
                    printf("Addition = %d", a+b);
                    break;

                case '2':
                    printf("Subtraction = %d", a-b);
                    break;

                case '3':
                    printf("Multiplication = %d", a*b);
                    break;

                case '4':
                    if(b!=0)
                        printf("Division = %d", a/b);
                    else
                        printf("Division not possible");
                    break;

                case '5':
                    if(b!=0)
                        printf("Modulo = %d", a%b);
                    else
                        printf("Modulo not possible");
                    break;
            }
        }
        else
        {
            printf("Invalid Choice");
        }
    }

    getch();
}

Sample Output

1. Addition
2. Subtraction
3. Multiplication
4. Division
5. Modulo
Enter choice or Q to quit: 1

Enter two numbers: 10 5
Addition = 15

Enter choice or Q to quit: Q

Question 24.

Enter a number from the user and check it is Armstrong number or not?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int num, temp, rem, count=0;
    int i, power, sum=0;

    clrscr();

    printf("Enter a number: ");
    scanf("%d", &num);

    temp=num;

    while(temp>0)
    {
        count++;
        temp=temp/10;
    }

    temp=num;

    while(temp>0)
    {
        rem=temp%10;
        power=1;

        for(i=1; i<=count; i++)
        {
            power=power*rem;
        }

        sum=sum+power;
        temp=temp/10;
    }

    if(sum==num)
        printf("Armstrong Number");
    else
        printf("Not an Armstrong Number");

    getch();
}

Sample Output

Enter a number: 153
Armstrong Number
