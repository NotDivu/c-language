Question 1

Enter an array from user and display all the values.

View Solution

C Program


#include <stdio.h>

int main()
{
    int a[5], i;

    printf("Enter 5 values:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Array values are:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output

Enter 5 values:
10 20 30 40 50
Array values are:
10 20 30 40 50

Question 2

View Solution

C Program

{

Question 10

Enter 5 values, then search a value in the array. Print Found or Not Found.

View Solution

C Program


#include <stdio.h>

int main()
{
    int a[5], i, num, found = 0;

    printf("Enter 5 values:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    printf("Enter value to search: ");
    scanf("%d", &num);

    for(i = 0; i < 5; i++)
    {
        if(a[i] == num)
        {
            found = 1;
            break;
        }
    }

    if(found == 1)
        printf("Found");
    else
        printf("Not Found");

    return 0;
}

Output

Enter 5 values:
10 20 30 40 50
Enter value to search: 30
Found

Question 11

Enter five values of array and sort in ascending order.

View Solution

C Program


#include <stdio.h>

int main()
{
    int a[5], i, j, temp;

    printf("Enter 5 values:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    for(i = 0; i < 5; i++)
    {
        for(j = i + 1; j < 5; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("Ascending order:\n");

    for(i = 0; i < 5; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output

Enter 5 values:
40 10 50 20 30
Ascending order:
10 20 30 40 50

Question 12

Find maximum value of array.

View Solution

C Program


#include <stdio.h>

int main()
{
    int a[5], i, max;

    printf("Enter 5 values:\n");

    for(i = 0; i < 5; i++)
    {
        scanf("%d", &a[i]);
    }

    max = a[0];

    for(i = 1; i < 5; i++)
    {
        if(a[i] > max)
        {
            max = a[i];
        }
    }

    printf("Maximum value = %d", max);

    return 0;
}

Output

Enter 5 values:
12 45 8 67 23
Maximum value = 67
