Question 1.

Write a program to print the following pattern on the screen:

11111
11111
11111
View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int i, j;

    clrscr();

    for(i=1; i<=3; i++)
    {
        for(j=1; j<=5; j++)
        {
            printf("1");
        }

        printf("\n");
    }

    getch();
}
