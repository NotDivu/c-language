2. Reverse Number

#include <stdio.h>
#include <conio.h>

int num;

void main()
{
    int rem,rev=0;

    clrscr();

    printf("Enter a number: ");
    scanf("%d",&num);

    while(num>0)
    {
        rem=num%10;
        rev=rev*10+rem;
        num=num/10;
    }

    printf("Reverse = %d",rev);

    getch();
}

Sample Output

Enter a number: 1234
Reverse = 4321

3. Palindrome Number

#include <stdio.h>
#include <conio.h>

int num;

void main()
{
    int rem,rev=0,temp;

    clrscr();

    printf("Enter a number: ");
    scanf("%d",&num);

    temp=num;

    while(num>0)
    {
        rem=num%10;
        rev=rev*10+rem;
        num=num/10;
    }

    if(temp==rev)
        printf("Palindrome number");
    else
        printf("Not a palindrome number");

    getch();
}

Sample Output

Enter a number: 121
Palindrome number

5. Armstrong Number

#include <stdio.h>
#include <conio.h>

int num;

void main()
{
    int rem,sum=0,temp;

    clrscr();

    printf("Enter a number: ");
    scanf("%d",&num);

    temp=num;

    while(num>0)
    {
        rem=num%10;
        sum=sum+rem*rem*rem;
        num=num/10;
    }

    if(temp==sum)
        printf("Armstrong number");
    else
        printf("Not an Armstrong number");

    getch();
}

Sample Output

Enter a number: 153
Armstrong number

Question 5.

Swap numbers using Call by Address?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void swap(int *a,int *b)
{
    int temp;

    temp=*a;
    *a=*b;
    *b=temp;
}

void main()
{
    int a,b;

    clrscr();

    printf("Enter two numbers: ");
    scanf("%d%d",&a,&b);

    swap(&a,&b);

    printf("After swapping:\n");
    printf("a = %d\n",a);
    printf("b = %d",b);

    getch();
}

Sample Output

Enter two numbers: 10 20

After swapping:
a = 20
b = 10

Question 7.

Make a function to find out the power of a number. Enter the number and power from the user?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

int power(int num,int p)
{
    int i,ans=1;

    for(i=1;i<=p;i++)
    {
        ans=ans*num;
    }

    return ans;
}

void main()
{
    int num,p,ans;

    clrscr();

    printf("Enter a number: ");
    scanf("%d",&num);

    printf("Enter power: ");
    scanf("%d",&p);

    ans=power(num,p);

    printf("Answer = %d",ans);

    getch();
}

Sample Output

Enter a number: 2
Enter power: 3
Answer = 8
