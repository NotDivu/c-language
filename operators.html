Question 1.

Write a program to show an example of pre-increment?

View Solution

#include <conio.h>

{

    clrscr();

}

Question 5.

Write a program to enter 2 numbers from user, add the numbers and show the answer, again take 2 numbers from user and subtract the numbers, display the answer, again take 2 numbers from user and multiply the numbers, display the answer, again take 2 numbers from user and divide the numbers, display the answer, again take 2 numbers, find the remainder and display the answer?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void main()
{
    int a,b;
    clrscr();

    printf("Enter two numbers for addition: ");
    scanf("%d%d",&a,&b);
    printf("Addition = %d\n",a+b);

    printf("\nEnter two numbers for subtraction: ");
    scanf("%d%d",&a,&b);
    printf("Subtraction = %d\n",a-b);

    printf("\nEnter two numbers for multiplication: ");
    scanf("%d%d",&a,&b);
    printf("Multiplication = %d\n",a*b);

    printf("\nEnter two numbers for division: ");
    scanf("%d%d",&a,&b);

    if(b!=0)
        printf("Division = %d\n",a/b);
    else
        printf("Division not possible\n");

    printf("\nEnter two numbers for remainder: ");
    scanf("%d%d",&a,&b);

    if(b!=0)
        printf("Remainder = %d",a%b);
    else
        printf("Remainder not possible");

    getch();
}

Sample Output

Enter two numbers for addition: 10 5
Addition = 15

Enter two numbers for subtraction: 20 5
Subtraction = 15

Enter two numbers for multiplication: 4 5
Multiplication = 20

Enter two numbers for division: 20 4
Division = 5

Enter two numbers for remainder: 10 3
Remainder = 1

Question 6.

Write a program to enter a number from the user. The length of the number depends on the user. Programmer should print the last digit of the number?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void main()
{
    int num,rem;
    clrscr();

    printf("Enter a number: ");
    scanf("%d",&num);

    rem=num%10;

    if(rem<0)
        rem=-rem;

    printf("Last digit = %d",rem);

    getch();
}

Sample Output

Enter a number: 12345
Last digit = 5

Question 11.

Question 12.

Calculate simple interest = (P x R x T) / 100?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void main()
{
    float p,r,t,si;
    clrscr();

    printf("Enter principal amount: ");
    scanf("%f",&p);

    printf("Enter rate: ");
    scanf("%f",&r);

    printf("Enter time: ");
    scanf("%f",&t);

    si=(p*r*t)/100;

    printf("Simple Interest = %f",si);

    getch();
}

Sample Output

Enter principal amount: 1000
Enter rate: 5
Enter time: 2
Simple Interest = 100.000000
