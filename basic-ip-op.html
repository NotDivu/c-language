Question 2.

Write a program to read a character from the keyboard and display it on the screen.

View Solution

C Program Code

#include <conio.h>

void main()
{

Question 3.

Write a program to read only 4 characters from the keyboard and display them on the screen in separate lines.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char ch;
    int i;

    clrscr();

    printf("Enter 4 characters:\n");

    for(i=0; i<4; i++)
    {
        scanf(" %c", &ch);
        printf("%c\n", ch);
    }

    getch();
}

Sample Output

Enter 4 characters:
A
B
C
D

A
B
C
D

Question 6.

Question 12.

Enter password and display it like ******?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char ch;
    int i=0;

    clrscr();

    printf("Enter password: ");

    while(1)
    {
        ch = getch();

        if(ch==13)
        {
            break;
        }

        printf("*");
        i++;
    }

    printf("\nPassword entered");

    getch();
}

Sample Output

Enter password: ******

Password entered

Question 13.

Enter two characters from user and print their ASCII values.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char ch1, ch2;

    clrscr();

    printf("Enter first character: ");
    scanf(" %c", &ch1);

    printf("Enter second character: ");
    scanf(" %c", &ch2);

    printf("ASCII value of %c = %d\n", ch1, ch1);
    printf("ASCII value of %c = %d", ch2, ch2);

    getch();
}

Sample Output

Enter first character: A
Enter second character: B

ASCII value of A = 65
ASCII value of B = 66

Question 14.

Print all ASCII characters from 0 to 127.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int i;

    clrscr();

    for(i=0; i<=127; i++)
    {
        printf("%d = %c\n", i, i);
    }

    getch();
}
