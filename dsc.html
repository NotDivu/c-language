Question 1.

Write a program to enter the number from the user and check it is positive or negative or 0 (zero)?

#include <conio.h>

{

    if(num>0)
        printf("Positive number");

    switch(days)
    {
        case 30:
            printf("April, June, September, November");

        case 31:
            printf("January, March, May, July, August, October, December");
            break;

        case 29:

            printf("Invalid input");

}

{

    scanf("%d%d%d",&date,&month,&year);

    if(month>=1 && month<=12)
    {
        if(month==2)
        {
            if(year%400==0 || (year%4==0 && year%100!=0))
                days=29;
            else
                days=28;

            days=30;
        else
            days=31;

        if(date>=1 && date<=days)
            printf("Valid date");
        else
            printf("Invalid date");
    }
    else
        printf("Invalid date");

}

Question 22.

Enter current date and DOB of user and find out the difference?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void main()
{
    int cd,cm,cy;
    int bd,bm,by;
    int days,months,years;
    clrscr();

    printf("Enter current date, month and year: ");
    scanf("%d%d%d",&cd,&cm,&cy);

    printf("Enter DOB date, month and year: ");
    scanf("%d%d%d",&bd,&bm,&by);

    if(cy<by || (cy==by && cm<bm) ||
       (cy==by && cm==bm && cd<bd))
    {
        printf("Invalid dates");
    }
    else
    {
        days=cd-bd;
        months=cm-bm;
        years=cy-by;

        if(days<0)
        {
            days=days+30;
            months--;
        }

        if(months<0)
        {
            months=months+12;
            years--;
        }

        printf("Difference:\n");
        printf("Years = %d\n",years);
        printf("Months = %d\n",months);
        printf("Days = %d",days);
    }

    getch();
}

Sample Output

Enter current date, month and year: 30 10 2026
Enter DOB date, month and year: 15 5 2006

Difference:
Years = 20
Months = 5
Days = 15

Question 23.

Write a program to calculate area and perimeter of rectangle?

View Solution

C Program Code

#include <stdio.h>
#include <conio.h>

void main()
{
    int length,width,area,perimeter;
    clrscr();

    printf("Enter length: ");
    scanf("%d",&length);

    printf("Enter width: ");
    scanf("%d",&width);

    area=length*width;
    perimeter=2*(length+width);

    printf("Area = %d\n",area);
    printf("Perimeter = %d",perimeter);

    getch();
}

Sample Output

Enter width: 5
Area = 50
Perimeter = 30
