Question 1.

Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

struct employee
{
    int id;
    char name[30];
    char designation[30];
};

void main()
{
    struct employee emp[50];
    int i;

    clrscr();

    for(i=0; i<50; i++)
    {
        printf("Enter ID, Name and Designation of Employee %d:\n", i+1);
        scanf("%d %s %s", &emp[i].id, emp[i].name, emp[i].designation);
    }

    printf("\nID\tName\tDesignation\n");

    for(i=0; i<50; i++)
    {
        printf("%d\t%s\t%s\n", emp[i].id, emp[i].name, emp[i].designation);
    }

    getch();
}

Sample Output

101 Rahul Manager

Enter ID, Name and Designation of Employee 2:
102 Aman Clerk

101     Rahul   Manager
102     Aman    Clerk

Question 3.

Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

struct employee
{
    int id;
    char name[30];
    char designation[30];
};

void main()
{
    struct employee emp[50];
    int i, j;

    clrscr();

    for(i=0; i<50; i++)
    {
        printf("Enter ID, Name and Designation:\n");
        scanf("%d %s %s", &emp[i].id, emp[i].name, emp[i].designation);
    }

    printf("\nID\tCharacter\n");

    for(i=0; i<50; i++)
    {
        for(j=0; emp[i].name[j]!='\0'; j++)
        {
            printf("%d\t%c\n", emp[i].id, emp[i].name[j]);
        }
    }

    getch();
}

Sample Output

Enter ID, Name and Designation:
101 Rahul Manager

ID      Character
101     R
101     a
101     h
101     u
101     l

Question 4.

Enter student details like roll, name, fee, DOB and print all the details by ascending order of roll number?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

struct student
{
    int roll;
    char name[30];
    float fee;
    char dob[15];
};

void main()
{
    struct student s[5], temp;
    int i, j;

    clrscr();

    for(i=0; i<5; i++)
    {
        printf("Enter Roll, Name, Fee and DOB:\n");
        scanf("%d %s %f %s", &s[i].roll, s[i].name, &s[i].fee, s[i].dob);
    }

    for(i=0; i<5; i++)
    {
        for(j=0; j<4-i; j++)
        {
            if(s[j].roll > s[j+1].roll)
            {
                temp = s[j];
                s[j] = s[j+1];
                s[j+1] = temp;
            }
        }
    }

    printf("\nRoll\tName\tFee\tDOB\n");

    for(i=0; i<5; i++)
    {
        printf("%d\t%s\t%.2f\t%s\n",



        s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }

    getch();
}
