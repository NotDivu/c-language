Question 1.

Enter a string from user and convert in uppercase?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char str[100];
    int i;

    clrscr();

    printf("Enter a string: ");
    scanf("%s", str);

    for(i=0; str[i]!='\0'; i++)
    {
        if(str[i]>='a' && str[i]<='z')
        {
            str[i] = str[i]-32;
        }
    }

    printf("Uppercase = %s", str);

    getch();
}

Sample Output

Enter a string: hello
Uppercase = HELLO

Question 3.

Enter a string from user and convert into sentence case?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char str[100];
    int i;

    clrscr();

    printf("Enter a string: ");
    scanf(" %[^\n]", str);

    for(i=0; str[i]!='\0'; i++)
    {
        if(i==0 && str[i]>='a' && str[i]<='z')
        {
            str[i] = str[i]-32;
        }
        else if(i!=0 && str[i]>='A' && str[i]<='Z')
        {
            str[i] = str[i]+32;
        }
    }

    printf("Sentence Case = %s", str);

    getch();
}

Sample Output

Enter a string: hELLO wORLD
Sentence Case = Hello world

Question 10.

Enter a string and a number. If number is 0 print first character, if 1 print second character, if 2 print third character, if 3 print fourth character and if out of string print out of bound?

{

Question 18.

String palindrome example (using reverse)?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    char str1[100], str2[100];
    int i, j=0, flag=0;

    clrscr();

    printf("Enter a string: ");
    scanf("%s", str1);

    for(i=0; str1[i]!='\0'; i++);

    for(i=i-1; i>=0; i--)
    {
        str2[j]=str1[i];
        j++;
    }

    str2[j]='\0';

    for(i=0; str1[i]!='\0' || str2[i]!='\0'; i++)
    {
        if(str1[i]!=str2[i])
        {
            flag=1;
            break;
        }
    }

    if(flag==0)
        printf("String is Palindrome");
    else
        printf("String is not Palindrome");

    getch();
}

Sample Output

Enter a string: madam
String is Palindrome

Question 19.

Enter the data from user. The data consists of roll number, name, class and marks of 5 subjects?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

struct student
{
    int roll;
    char name[30];
    char class[20];
    int marks[5];
};

void main()
{
    struct student s;
    int i;

    clrscr();

    printf("Enter Roll Number: ");
    scanf("%d", &s.roll);

    printf("Enter Name: ");
    scanf("%s", s.name);

    printf("Enter Class: ");
    scanf("%s", s.class);

    for(i=0; i<5; i++)
    {
        printf("Enter marks of subject %d: ", i+1);
        scanf("%d", &s.marks[i]);
    }

    printf("\nStudent Details\n");

    printf("Roll Number = %d\n", s.roll);
    printf("Name = %s\n", s.name);
    printf("Class = %s\n", s.class);

    printf("Marks of 5 Subjects:\n");

    for(i=0; i<5; i++)
    {
        printf("Subject %d = %d\n", i+1, s.marks[i]);
    }

    getch();
}

Enter Name: Rahul
Enter Class: BCA

Enter marks of subject 3: 85
Enter marks of subject 4: 90
Enter marks of subject 5: 70

Student Details

Name = Rahul
Class = BCA

Marks of 5 Subjects:
Subject 1 = 80
