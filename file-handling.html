File Handling

Question 1.

Enter the file name from user, enter the data from the user and put the data in the file, but data should be in capital letters.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    FILE *fp;
    char fname[30], ch;

    clrscr();

    printf("Enter file name: ");
    scanf("%s", fname);

    fp = fopen(fname, "w");

    printf("Enter data (# to stop): ");
    getchar();

    while ((ch = getchar()) != '#')
    {
        if (ch >= 'a' && ch <= 'z')
        {
            ch = ch - 32;
        }

        fputc(ch, fp);
    }

    fclose(fp);

    printf("Data saved successfully.");

    getch();
}

Sample Output

Enter file name: data.txt
Enter data (# to stop): hello world#
Data saved successfully.

File Content:
HELLO WORLD

Question 4.

Create a file and copy the file into another file.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    FILE *f1, *f2;
    char ch;

    clrscr();

    f1 = fopen("first.txt", "w");

    printf("Enter data (# to stop): ");

    while ((ch = getchar()) != '#')
    {
        fputc(ch, f1);
    }

    fputc('#', f1);
    fclose(f1);

    f1 = fopen("first.txt", "r");
    f2 = fopen("second.txt", "w");

    ch = fgetc(f1);

    while (ch != '#')
    {
        fputc(ch, f2);
        ch = fgetc(f1);
    }

    fclose(f1);
    fclose(f2);

    printf("File copied successfully.");

    getch();
}

Sample Output

Enter data (# to stop): Hello World#
File copied successfully.

Second File Content:
Hello World

Question 5.

Create 2 files and concatenate them into a third file.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    FILE *f1, *f2, *f3;
    char ch;

    clrscr();

    f1 = fopen("first.txt", "w");

    printf("Enter first file data (# to stop): ");

    while ((ch = getchar()) != '#')
    {
        fputc(ch, f1);
    }

    fputc('#', f1);
    fclose(f1);

    f2 = fopen("second.txt", "w");

    printf("Enter second file data (# to stop): ");

    while ((ch = getchar()) != '#')
    {
        fputc(ch, f2);
    }

    fputc('#', f2);
    fclose(f2);

    f1 = fopen("first.txt", "r");
    f2 = fopen("second.txt", "r");
    f3 = fopen("third.txt", "w");

    ch = fgetc(f1);

    while (ch != '#')
    {
        fputc(ch, f3);
        ch = fgetc(f1);
    }

    fputc(' ', f3);

    ch = fgetc(f2);

    while (ch != '#')
    {
        fputc(ch, f3);
        ch = fgetc(f2);
    }

    fclose(f1);
    fclose(f2);
    fclose(f3);

    printf("Files concatenated successfully.");

    getch();
}

Sample Output

Enter first file data (# to stop): Hello#
Enter second file data (# to stop): World#
Files concatenated successfully.

Third File Content:
Hello World

Question 6.

Create a file and count the number of characters in the file.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    FILE *fp;
    char ch;
    int count = 0;

    clrscr();

    fp = fopen("data.txt", "w");

    printf("Enter data (# to stop): ");

    while ((ch = getchar()) != '#')
    {
        fputc(ch, fp);
    }

    fputc('#', fp);
    fclose(fp);

    fp = fopen("data.txt", "r");

    ch = fgetc(fp);

    while (ch != '#')
    {
        count++;
        ch = fgetc(fp);
    }

    fclose(fp);

    printf("Total characters = %d", count);

    getch();
}

Sample Output

Enter data (# to stop): Hello World#
Total characters = 11

Question 7.

Create a file and count the number of words in the file.

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    FILE *fp;
    char ch;
    int count = 0, flag = 0;

    clrscr();

    fp = fopen("data.txt", "w");

    printf("Enter data (# to stop): ");

    while ((ch = getchar()) != '#')
    {
        fputc(ch, fp);
    }

    fputc('#', fp);
    fclose(fp);

    fp = fopen("data.txt", "r");

    ch = fgetc(fp);

    while (ch != '#')
    {
        if (ch == ' ' || ch == '\n' || ch == '\t')
        {
            flag = 0;
        }
        else if (flag == 0)
        {
            count++;
            flag = 1;
        }

        ch = fgetc(fp);
    }

    fclose(fp);

    printf("Total words = %d", count);

    getch();
}

Sample Output

Enter data (# to stop): Hello World#
Total words = 2
