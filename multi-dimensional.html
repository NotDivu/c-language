Multi Dimensional Array

Question 1.

Enter the four numbers from user and print it into matrix?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a[2][2];
    int i, j;

    clrscr();

    printf("Enter four numbers:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Matrix is:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            printf("%d\t", a[i][j]);
        }
        printf("\n");
    }

    getch();
}

Sample Output

Enter four numbers:
1 2
3 4

Matrix is:
1       2
3       4

Question 3.

Enter two matrices from user and add them?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a[2][2], b[2][2], c[2][2];
    int i, j;

    clrscr();

    printf("Enter first matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            c[i][j] = a[i][j] + b[i][j];
        }
    }

    printf("Addition of Matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            printf("%d\t", c[i][j]);
        }
        printf("\n");
    }

    getch();
}

Question 5.

Enter two matrices from user and multiply them?

View Solution

C Program Code


#include <stdio.h>
#include <conio.h>

void main()
{
    int a[2][2], b[2][2], c[2][2];
    int i, j, k;

    clrscr();

    printf("Enter first matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            scanf("%d", &a[i][j]);
        }
    }

    printf("Enter second matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            scanf("%d", &b[i][j]);
        }
    }

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            c[i][j] = 0;

            for(k=0; k<2; k++)
            {
                c[i][j] = c[i][j] + a[i][k] * b[k][j];
            }
        }
    }

    printf("Multiplication of Matrix:\n");

    for(i=0; i<2; i++)
    {
        for(j=0; j<2; j++)
        {
            printf("%d\t", c[i][j]);
        }
        printf("\n");
    }

    getch();
}

Sample Output

Enter first matrix:
1 2
3 4

Enter second matrix:
5 6
7 8

Multiplication of Matrix:
19      22
43      50
